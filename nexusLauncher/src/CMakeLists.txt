cmake_minimum_required(VERSION 3.10)
project(nexus_launcher CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(IMGUI third_party/imgui.cpp third_party/imgui_draw.cpp third_party/imgui_tables.cpp third_party/imgui_widgets.cpp third_party/imgui_stdlib.cpp)
if(WIN32)
  add_executable(nexus_launcher WIN32 app.cpp plat_win32.cpp ${IMGUI})
  target_link_libraries(nexus_launcher ws2_32 shell32 gdi32 comdlg32 imm32 ole32)
  if(MINGW)
    target_link_options(nexus_launcher PRIVATE -static)
  endif()
else()
  add_executable(nexus_launcher app.cpp plat_x11.cpp ${IMGUI})
  find_package(Threads REQUIRED)
  target_link_libraries(nexus_launcher Threads::Threads ${CMAKE_DL_LIBS})
endif()
target_include_directories(nexus_launcher PRIVATE third_party)
